% Editable vector framework figure for TrustKAN-Climate.
%
% This diagram is the design that was evaluated after the two repairs: a
% causal dilated stem whose reach is a closed-form property of its depth, an
% attention readout, and a reliability fusion whose weight is selected on the
% calibration partition. Components the evidence refutes --- the KAN layer's
% interpretability, and the claim that the shift term earns fusion mass ---
% are drawn hatched, so the figure cannot outrun the results.
% The picture is scaled to \textwidth, so the readable size of a label is its
% font size times that scale. Node widths set the natural width and therefore
% the scale; a larger font only adds line wraps, making the boxes taller without
% shrinking anything further. That is why the font here is deliberately large.
\begin{tikzpicture}[
  font=\large,
  node distance=6mm and 7mm,
  box/.style={draw, rounded corners=2pt, align=center, minimum height=13mm, text width=24mm, fill=white},
  data/.style={box, fill=blue!6, draw=blue!55!black},
  model/.style={box, fill=orange!8, draw=orange!70!black},
  trust/.style={box, fill=green!7, draw=green!55!black},
  decision/.style={box, fill=purple!7, draw=purple!60!black},
  gone/.style={densely dashed, draw=black!45, fill=black!8, text=black!65},
  arr/.style={-{Latex[length=2mm]}, thick, draw=black!65},
  faded/.style={arr, densely dashed, draw=black!35}
]
\node[data] (data) {Climate / weather\\history $\mathbf{X}_t$\\[1pt]\scriptsize $\robustHistory$ steps};
\node[model, right=of data] (temporal) {Causal dilated\\stem $g_\theta$\\[1pt]\scriptsize reads $\robustFieldTrust$ of $\robustHistory$};
\node[model, right=of temporal] (kan) {KAN functional\\representation};
\node[model, right=of kan] (readout) {Attention\\readout};
\node[model, right=of readout] (embed) {Embedding\\$\mathbf e_t$};

\node[model, above right=8mm and 4mm of embed] (point) {Point head\\$\hat{\mathbf y}_t$};
\node[model, right=of embed] (quant) {Quantile head\\$\hat q^{(\tau)}$};
\node[trust, right=of quant] (conf) {Static / rolling /\\adaptive conformal};

\node[trust, below=of embed] (shift) {Representation\\shift $d_t$};
\node[trust, right=of conf] (width) {Interval-width\\reliability};
\node[trust, below=of width] (fusion) {Calibration-selected\\fusion $r_t$};
\node[decision, right=of fusion] (gate) {Selective gate\\accept / abstain};
\node[decision, right=of gate] (eval) {Risk--coverage\\decision support};

\draw[arr] (data) -- (temporal);
\draw[arr] (temporal) -- (kan);
\draw[arr] (kan) -- (readout);
\draw[arr] (readout) -- (embed);
\draw[arr] (embed) -- (point);
\draw[arr] (embed) -- (quant);
\draw[arr] (quant) -- (conf);
\draw[arr] (conf) -- (width);
\draw[arr] (embed) -- (shift);
\draw[arr] (width) -- (fusion);
\draw[arr] (shift.east) -| (fusion.south);
\draw[arr] (fusion) -- (gate);
\draw[arr] (point.east) -| (gate.north);
\draw[arr] (gate) -- (eval);

\node[below=14mm of kan, align=center, text width=76mm, draw=black!45, densely dashed,
      rounded corners=2pt, fill=black!8, text=black!65, inner sep=4pt] (interp)
  {\textbf{Interpretability analysis:} univariate maps over latent channels;
   index-matched, shuffled, and within-seed controls all refute stability};
\draw[faded] (kan.south) -- (interp.north);

\node[below=5mm of interp, align=center, text width=100mm] (legend)
  {Solid components are those under test. Hatched: the interpretability claim
   is refuted; the shift term remains in the fusion and the data may withhold
   mass from it, which is the point of selecting $\omega$ on calibration.};
\end{tikzpicture}
